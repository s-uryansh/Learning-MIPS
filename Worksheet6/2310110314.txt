============================================

Design:
{
module full_adder_s(
  input a, b, cin,
  output sum, carry
);
  
  wire w1, w2, w3;
  
  xor (w1, a, b);
  xor (sum, w1, cin);
  
  and (w2, a, b);
  and (w3, w1, cin);
  or (carry, w2, w3);
endmodule
}

============================================
Design:

module mux_4_to_1(
    input a0, a1, a2, a3,
    input selectline0, selectline1,
    output y
);
    wire sel0_n, sel1_n;
    wire y0, y1, y2, y3;

    not(sel0_n, selectline0);
    not(sel1_n, selectline1);

    and(y0, a0, sel0_n, sel1_n);
    and(y1, a1, sel0_n, selectline1);
    and(y2, a2, selectline0, sel1_n);
    and(y3, a3, selectline0, selectline1);

    or(y, y0, y1, y2, y3);
endmodule
